# This file contains a mission, only offered on Omnis planets, that lets you enable any paint at no cost

color "paint job: selected" .5 .875 .5 0.
color "paint job: unselected" .25 .75 .25 0.
color "paint job: unavailable" .125 .625 .125 0.

# Government stubs to prevent errors if Omnis is not installed.
government "Omnis"
government "Faction"
government "Arena"

mission "Interstellar Paint: Switchboard"
	name "Change Paint"
	description `This world is home to a mysterious vendor of starship paint known for being able to provide any kind of paint imagineable on short notice. Press "Accept" to visit.`
	color selected "paint job: selected"
	color unselected "paint job: unselected"
	color unavailable "paint job: unavailable"
	job
	repeat
	source
		government "Omnis" "Faction" "Arena"
	to fail
		has "Fail Paint Change"
	on fail
		clear "Fail Paint Change"
	on accept
		set "Fail Paint Change"
		conversation
			`Arriving at the paint vendor, you find only a single ancient touchscreen watched over by a million tiny cameras.`
			label menu_main
			`	[USER WILL PLEASE SELECT PAINT TYPE], reads the menu on the touchscreen.`
			choice
				`	[Unrestricted]`
					goto menu_unrestricted
				`	[Outlaw]`
					goto menu_outlaw
				`	[Story-Gated]`
					goto menu_story
				`	[Reputation-Gated]`
					goto menu_reputation
				`	[Traditionally Unobtainable]`
					goto menu_unobtainable
				`	[Exit]`
			
			`	You press the button labelled "exit" and the screen goes black. Looks like you're not getting a new paint job today after all.`
