# This file contains custom swizzles that can be applied with certain paints

# teal and purple markings
swizzle "total inversion"
	red   .0 .5 .5 0
	green .5 .0 .5 0
	blue  .5 .5 .0 0
	alpha .0 .0 .0 1

# extremely varied blue markings
swizzle "magenta inversion"
	red   .0 .5 .5 0
	green .0 1. .0 0
	blue  .5 .5 .0 0
	alpha .0 .0 .0 1

# extremely varied green markings
swizzle "red inversion"
	red   .0 .5 .5 0
	green .0 1. .0 0
	blue  .0 .0 1. 0
	alpha .0 .0 .0 1

# green and brown markings
swizzle "yellow inversion"
	red   .0 .5 .5 0
	green .5 0 .5 0
	blue  .0 .0 1. 0
	alpha .0 .0 .0 1

# uniform orange markings
swizzle "green inversion"
	red   .0 .0 1. 0
	green .5 .0 .5 0
	blue  .0 .0 1. 0
	alpha .0 .0 .0 1

# light pink and purple markings
swizzle "cyan inversion"
	red   1. .0 0 0
	green .5 .0 .5 0
	blue  .5 .5 .0 0
	alpha .0 .0 .0 1

# extremely varied pink markings
swizzle "blue inversion"
	red   1. .0 .0 0
	green .0 1. .0 0
	blue  .5 .5 .0 0
	alpha .0 .0 .0 1

# a proper inversion, just for fun
swizzle "ghostpaint"
	red   -1 0 0 1
	green 0 -1 0 1
	blue  0 0 -1 1
	alpha 0 0 0 1
